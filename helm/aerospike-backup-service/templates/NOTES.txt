{{- $ctx := include "aerospike-backup-service.contextPath" . -}}
{{- $scheme := include "aerospike-backup-service.primaryListener" . -}}
{{- $svcPort := include "aerospike-backup-service.primaryServicePort" . -}}
1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  {{- /* A catch-all "/" path forwards the full URL, so the API sits under the context path. */}}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ if eq .path "/" }}{{ $ctx }}{{ else }}{{ .path }}{{ end }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[?(@.name=='{{ $scheme }}')].nodePort}" services {{ include "aerospike-backup-service.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo {{ $scheme }}://$NODE_IP:$NODE_PORT{{ $ctx }}
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch its status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "aerospike-backup-service.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "aerospike-backup-service.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo {{ $scheme }}://$SERVICE_IP:{{ $svcPort }}{{ $ctx }}
{{- else if contains "ClusterIP" .Values.service.type }}
  {{- /* Forward to the Service, not a pod: the finished `helm test` pod carries the same labels. */}}
  {{- $local := ternary 8080 8443 (eq $scheme "http") }}
  echo "Visit {{ $scheme }}://127.0.0.1:{{ $local }}{{ $ctx }} to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "aerospike-backup-service.fullname" . }} {{ $local }}:{{ $svcPort }}
{{- end }}

2. Endpoints under that URL:
  health     liveness
  ready      readiness
  metrics    Prometheus metrics
  api-docs/  API documentation
  v1/        REST API
